cmake_minimum_required(VERSION 3.24)
project(uxzViewTools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ZXING_READERS ON CACHE BOOL "" FORCE)
set(ZXING_WRITERS OFF CACHE STRING "" FORCE)
set(ZXING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZXING_C_API OFF CACHE BOOL "" FORCE)
set(ZXING_DEPENDENCIES LOCAL CACHE STRING "" FORCE)
# Avoid mixing optimization levels with a precompiled standard-library header.
set(ZXING_DISABLE_PCH ON CACHE BOOL "" FORCE)
add_subdirectory("${ZXING_SOURCE}" zxing)
if(MINGW AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 16)
    # GCC 16/MinGW emits unresolved std::string move constructors at -Os.
    # Override only upstream's size-optimized units, keeping its -O3 reader code.
    get_target_property(zxing_sources ZXing SOURCES)
    foreach(source IN LISTS zxing_sources)
        if(NOT IS_ABSOLUTE "${source}")
            set(source "${ZXING_SOURCE}/core/${source}")
        endif()
        get_source_file_property(flags "${source}" DIRECTORY "${ZXING_SOURCE}/core" COMPILE_FLAGS)
        if(flags STREQUAL "-Os")
            set_source_files_properties("${source}" DIRECTORY "${ZXING_SOURCE}/core" PROPERTIES COMPILE_FLAGS "-O2")
        endif()
    endforeach()
endif()
add_executable(uxz-code-reader code_reader.cpp)
target_link_libraries(uxz-code-reader PRIVATE ZXing::ZXing)
install(TARGETS uxz-code-reader RUNTIME DESTINATION bin)
